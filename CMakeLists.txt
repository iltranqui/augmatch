# CpuStage always operates on host views. CUDA builds select device versions
# for the public transform symbols, so retain the exact existing CPU sources
# under a private namespace for Pipeline's host adapters.
add_library(augmatch_pipeline_host_primitives OBJECT EXCLUDE_FROM_ALL
  src/hsv_cpu.cpp src/color_cpu.cpp src/filter_cpu.cpp src/transforms_cpu.cpp)
target_include_directories(augmatch_pipeline_host_primitives PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(augmatch_pipeline_host_primitives PRIVATE
  augmatch=augmatch_pipeline_host AUGMATCH_HAS_CUDA=0)
set_target_properties(augmatch_pipeline_host_primitives PROPERTIES
  CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
