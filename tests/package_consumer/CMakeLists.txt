cmake_minimum_required(VERSION 3.24)
project(augmatch_package_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(augmatch CONFIG REQUIRED)
if(augmatch_HAS_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
endif()
add_executable(augmatch_package_consumer main.cpp)
target_link_libraries(augmatch_package_consumer PRIVATE augmatch::augmatch)
if(augmatch_HAS_CUDA)
  set_target_properties(augmatch_package_consumer PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
endif()
